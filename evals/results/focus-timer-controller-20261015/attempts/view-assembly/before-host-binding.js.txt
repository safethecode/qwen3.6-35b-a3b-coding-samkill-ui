function currentView(){const state=store.getState();if(!state.readReady){return{readReady:false,session:null,error:state.error,fieldInvalid:fieldInvalid,configOpen:configOpen}}let session;try{session=snapshot(state.record,Date.now())}catch(e){if(e.message==="INVALID_CLOCK"){return{readReady:false,session:null,error:e.message,fieldInvalid:fieldInvalid,configOpen:configOpen}}throw e}return{readReady:true,session:session,error:state.error,fieldInvalid:fieldInvalid,configOpen:configOpen}}

function refresh() {
  const view = currentView();
  renderTimer(view);
  return view;
}

function returnFocus() {
  if (configure && configure.offsetParent !== null && !configure.disabled) {
    configure.focus();
    return;
  }
  if (primary && primary.offsetParent !== null && !primary.disabled) {
    primary.focus();
    return;
  }
  if (mainReload && mainReload.offsetParent !== null && !mainReload.disabled) {
    mainReload.focus();
    return;
  }
}
