function openConfig() { const view = currentView(); if (view.readReady || view.session.phase !== 'ready' && view.session.phase !== 'completed') return; fieldInvalid = false; document.getElementById('duration-input').value = String(view.config.durationMinutes); document.getElementById('duration-dialog').showModal(); refresh(); document.getElementById('duration-input').focus(); } function closeConfig(event) { if (event) event.preventDefault(); fieldInvalid = false; document.getElementById('duration-dialog').close(); refresh(); returnFocus(); } function submitConfig(event) { event.preventDefault(); const inputEl = document.getElementById('duration-input'); const parsed = parseDuration(inputEl.value); if (!parsed.ok) { fieldInvalid = true; refresh(); inputEl.focus(); return; } fieldInvalid = false; const result = store.transact({ type: 'CONFIGURE', minutes: parsed.minutes }, Date.now()); if (result.ok) closeConfig(); else refresh(); } function inputConfig() { fieldInvalid = false; refresh(); }
