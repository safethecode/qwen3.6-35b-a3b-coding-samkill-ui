function currentView() {
  try {
    const { record, error } = store.getState();
    if (!record) return { readReady: false, session: null, error: error || 'NO_RECORD', fieldInvalid: fieldInvalid, configOpen: dialog.open };
    const snapshot = snapshot(record, Date.now());
    return {
      readReady: true,
      session: snapshot,
      error: error || null,
      fieldInvalid: fieldInvalid,
      configOpen: dialog.open
    };
  } catch (e) {
    if (e.message === 'INVALID_CLOCK') {
      return { readReady: false, session: null, error: 'INVALID_CLOCK', fieldInvalid: fieldInvalid, configOpen: dialog.open };
    }
    throw e;
  }
}

function refresh() {
  const view = currentView();
  renderTimer(view);
  return view;
}

function returnFocus() {
  if (configure && configure.offsetParent !== null && !configure.disabled) {
    configure.focus();
    return;
  }
  if (primary && primary.offsetParent !== null && !primary.disabled) {
    primary.focus();
    return;
  }
  if (mainReload && mainReload.offsetParent !== null && !mainReload.disabled) {
    mainReload.focus();
    return;
  }
}

function reloadSession(event) {
  const { ok, error } = store.reload();
  const view = refresh();
  if (ok) {
    if (dialog.open) {
      if (apply && apply.offsetParent !== null && !apply.disabled) {
        apply.focus();
      } else if (cancel && cancel.offsetParent !== null && !cancel.disabled) {
        cancel.focus();
      }
    } else {
      if (primary && primary.offsetParent !== null && !primary.disabled) {
        primary.focus();
      } else if (mainReload && mainReload.offsetParent !== null && !mainReload.disabled) {
        mainReload.focus();
      }
    }
  } else {
    returnFocus();
  }
}
