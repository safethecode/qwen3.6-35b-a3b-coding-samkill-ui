import{readFile,writeFile,mkdir}from'node:fs/promises';import{createServer}from'node:http';import{resolve,extname}from'node:path';import assert from'node:assert/strict';import{chromium}from'playwright';import{sourceBinding}from'../../scripts/source-binding.mjs';
const[target,out]=process.argv.slice(2);await mkdir(out,{recursive:true});const sourceHashes=await sourceBinding(target),server=createServer(async(req,res)=>{try{const p=new URL(req.url,'http://localhost').pathname.slice(1)||'index.html';if(!Object.hasOwn(sourceHashes,p))throw Error();res.setHeader('content-type',({'.html':'text/html;charset=utf-8','.js':'text/javascript','.css':'text/css','.svg':'image/svg+xml'})[extname(p)]||'application/octet-stream');res.end(await readFile(resolve(target,p)));}catch{res.writeHead(404).end();}});await new Promise(r=>server.listen(0,'127.0.0.1',r));const browser=await chromium.launch({executablePath:process.env.CHROME_PATH,headless:true});
try{const context=await browser.newContext({viewport:{width:390,height:900},timezoneId:'Asia/Seoul'}),page=await context.newPage();await page.goto(`http://127.0.0.1:${server.address().port}`);await page.evaluate(()=>document.fonts.ready);for(const[title,category]of[['알고리즘 두 문제 풀기','study'],['자료 구조 복습하기','study'],['산책 20분','life'],['내일 사용할 장보기 목록 정리하기','life']]){await page.locator(`[data-add-category=${category}]`).click();await page.locator('#task-name').fill(title);await page.locator('#save-task').click();}await page.locator('.task-check').first().check();assert.equal(await page.locator('#edit-tasks').getAttribute('aria-pressed'),'false');await page.screenshot({path:`${out}/390-reading.png`,fullPage:true});await page.locator('#edit-tasks').click();assert.equal(await page.locator('#edit-tasks').getAttribute('aria-pressed'),'true');await page.screenshot({path:`${out}/390-editing.png`,fullPage:true});await page.locator('[data-add-category=study]').click();await page.locator('#task-name').fill('정리할 내용');await page.screenshot({path:`${out}/390-form.png`,fullPage:true});assert.deepEqual(await sourceBinding(target),sourceHashes);await writeFile(`${out}/binding.json`,JSON.stringify({status:'DIAGNOSTIC_RENDER_ONLY_NOT_APPROVAL',sourceHashes,fixture:'Four realistic study/life tasks created through actual UI; one completed. Native edit toggle and create form. No product seeds or demo copy added.'},null,2));console.log('Captured realistic reading/editing/form screenshots; no quality approval');}finally{await browser.close();await new Promise(r=>server.close(r));}
