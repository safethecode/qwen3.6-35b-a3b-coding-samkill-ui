import { readFile, writeFile, mkdir } from 'node:fs/promises';
import { createHash } from 'node:crypto';
import { execFile } from 'node:child_process';
import { promisify } from 'node:util';
import { parse } from 'acorn';
import assert from 'node:assert/strict';
import { sourceBinding } from '../../../scripts/source-binding.mjs';
import { harnessBinding } from '../../../scripts/harness-binding.mjs';
import { inferenceFetch } from '../../../scripts/inference-http.mjs';
const root = 'runs/stay-booking-transfer-20261016/booking', target = root + '/source', unit = process.argv[2], run = promisify(execFile), hash = value => createHash('sha256').update(value).digest('hex');
const config = { quote: { file: 'stored-quote.js', checker: 'check-stored-quote.mjs', diagnosis: 'Padded source IDs such as stayId=" x " incorrectly return true. Contract demands strict already-trimmed identity, not validation of a trimmed temporary. Keep all other passed behaviors. Repair only ID validation guards, not date/price logic or imports.', section: '## isValidStoredQuote', next: '## isValidRequestStore' } }[unit];
assert.ok(config);
const out = root + '/repair-' + unit + '-identity-slots';
await mkdir(out);
const state = async (status, extra = {}) => { await writeFile(out + '/state.json', JSON.stringify({ status, pid: process.pid, at: new Date().toISOString(), ...extra }, null, 2)); console.log(status); };
try {
  const before = await readFile(target + '/' + config.file, 'utf8'), full = await readFile(root + '/STORED-DATA-CONTRACT.md', 'utf8'), contract = full.slice(full.indexOf(config.section), full.indexOf(config.next)), checker = await readFile(root + '/' + config.checker), driver = await readFile(new URL(import.meta.url)), sourceHashes = await sourceBinding(target), harnessHashes = await harnessBinding();
  const guidePath = 'skills/qwen-samkill-ui/references/decomposed-layout.md', fullGuide = await readFile(guidePath, 'utf8'), guide = fullGuide.slice(fullGuide.indexOf('## Behavior units'), fullGuide.indexOf('## Executable layout'));
  const old = before.slice(before.indexOf('  if (s1.length'), before.indexOf('  if (typeof value.start'));
  assert.equal(old.trim().split('\n').length, 3);
  const request = { model: 'qwen3.6:35b-a3b-coding', stream: false, think: false, options: { num_ctx: 16384, num_gpu: 16, num_batch: 128, num_predict: 512, temperature: .7, top_p: .8, top_k: 20, min_p: 0, presence_penalty: 1.5, repeat_penalty: 1 }, format: { type: 'object', properties: { new: { type: 'string' } }, required: ['new'], additionalProperties: false }, messages: [{ role: 'system', content: 'Return exactly JSON {"new":"three replacement if guard lines"}. Only3if statements, each returns false. Existing s1,s2,s3 are trimmed IDs and value.stayId/value.roomId/value.rateId are original strings. Preserve empty/length guards and also enforce strict identity. Do not introduce helper functions, return true, variable declarations, comments or Markdown. The host replaces only the3existing if lines.' }, { role: 'user', content: 'APPLICABLE BOUNDED GUIDE\n' + guide + '\nCONTRACT\n' + contract + '\nACTUAL FAILED ASSERTION AND DIAGNOSIS\n' + config.diagnosis + '\nREAD-ONLY MODULE CONTEXT\n' + before + '\nONLY WRITABLE3GUARD SLOT\n' + old }] };
  await writeFile(out + '/before.js.txt', before);
  await writeFile(out + '/driver.mjs.txt', driver);
  await writeFile(out + '/request.json', JSON.stringify({ request, sourceHashes, harnessHashes, checkerSha256: hash(checker), contractSha256: hash(full), driverSha256: hash(driver), guide: { path: guidePath, sourceSha256: hash(fullGuide), deliveredSha256: hash(guide) } }, null, 2));
  await run('pwsh.exe', ['-NoProfile', '-Command', `$c=@(Get-NetTCPConnection -State Established -ErrorAction SilentlyContinue | Where-Object {$_.RemotePort -eq 11434 -and $_.OwningProcess -ne ${process.pid}});$w=@(Get-CimInstance Win32_Process | Where-Object {$_.Name -in @('node.exe','python.exe','bun.exe') -and $_.ProcessId -ne ${process.pid} -and $_.CommandLine -match '(generate-service|generate-components|repair-component|(focus-timer|stay-booking).*(run-|repair-).*mjs)'});if($c.Count -or $w.Count){exit 9}`], { windowsHide: true, timeout: 15000 });
  await state('LOCAL_BOUNDED_IDENTITY_REPAIR_ACTIVE');
  const started = Date.now(), response = await inferenceFetch('http://127.0.0.1:11434/api/chat', { method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify(request), signal: AbortSignal.timeout(600000) });
  assert.equal(response.ok, true);
  const raw = await response.json();
  await writeFile(out + '/response.json', JSON.stringify({ elapsedMs: Date.now() - started, ...raw }, null, 2));
  assert.equal(raw.done_reason, 'stop');
  const patch = { ...JSON.parse(raw.message.content), old };
  assert.deepEqual(Object.keys(patch).sort(), ['new', 'old']);
  assert.equal(typeof patch.old, 'string'); assert.equal(typeof patch.new, 'string');
  assert.ok(patch.old.length && before.split(patch.old).length === 2 && patch.old !== patch.new && patch.old.split('\n').length + patch.new.split('\n').length < 20);
  assert.ok(!/\b(?:export|import|function|return true)\b|\/\/|\/\*/.test(patch.new));
  const guards = parse(patch.new, { ecmaVersion: 'latest', allowReturnOutsideFunction: true });
  assert.equal(guards.body.length, 3);
  assert.ok(guards.body.every(node => node.type === 'IfStatement' && node.consequent.type === 'ReturnStatement' && node.consequent.argument.value === false));
  const after = before.replace(patch.old, patch.new);
  parse(after, { ecmaVersion: 'latest', sourceType: 'module' });
  assert.deepEqual(await sourceBinding(target), sourceHashes); assert.deepEqual(await harnessBinding(), harnessHashes);
  assert.equal(hash(await readFile(root + '/' + config.checker)), hash(checker)); assert.equal(hash(await readFile(root + '/STORED-DATA-CONTRACT.md')), hash(full)); assert.equal(hash(await readFile(new URL(import.meta.url))), hash(driver)); assert.equal(hash(await readFile(guidePath)), hash(fullGuide));
  await writeFile(target + '/' + config.file, after);
  await writeFile(out + '/after.js.txt', after);
  let exitCode = 0, log;
  try { const checked = await run(process.execPath, [root + '/' + config.checker, target, out + '/checks'], { windowsHide: true, timeout: 30000 }); log = checked.stdout + checked.stderr; } catch (error) { exitCode = error.code || 1; log = error.stdout + error.stderr; }
  await writeFile(out + '/check.log', log);
  const result = { status: exitCode ? 'FAILED_LOCAL_REPAIR_PRESERVED' : 'SCOPED_LOCAL_REPAIR_ACCEPTED_NOT_PRODUCT_APPROVAL', exitCode, sourceHashes: await sourceBinding(target), harnessHashes, moduleSha256: hash(after), driverSha256: hash(driver) };
  await writeFile(out + '/result.json', JSON.stringify(result, null, 2));
  if (!exitCode) await writeFile(out + '/accepted.json', JSON.stringify(result, null, 2));
  await state(result.status, { exitCode }); process.exitCode = exitCode;
} catch (error) { await state('FAILED_REQUEST_OR_VALIDATION', { error: String(error.message).replaceAll(process.cwd(), 'REPOSITORY').slice(0, 1500) }); process.exitCode = 1; }
