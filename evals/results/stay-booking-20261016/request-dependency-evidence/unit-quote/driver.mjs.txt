import { readFile, writeFile, mkdir } from 'node:fs/promises';
import { createHash } from 'node:crypto';
import { execFile } from 'node:child_process';
import { promisify } from 'node:util';
import { parse } from 'acorn';
import assert from 'node:assert/strict';
import { sourceBinding } from '../../../scripts/source-binding.mjs';
import { harnessBinding } from '../../../scripts/harness-binding.mjs';
import { readElementDataContext } from '../../../scripts/element-data.mjs';
import { inferenceFetch } from '../../../scripts/inference-http.mjs';
import { removeComments } from '../../../scripts/comments.mjs';
const root = 'runs/stay-booking-transfer-20261016/booking', target = root + '/source', unit = process.argv[2], hash = x => createHash('sha256').update(x).digest('hex'), run = promisify(execFile);
const config = { quote: { filename: 'stored-quote.js', name: 'isValidStoredQuote', checker: 'check-stored-quote.mjs', section: '## isValidStoredQuote', next: '## isValidRequestStore', pointer: '/quote', imports: ['./dates.js'], accepted: ['dates.js'], tokens: 2048 }, store: { filename: 'request-store.js', name: 'isValidRequestStore', checker: 'check-request-store.mjs', section: '## isValidRequestStore', next: '## applyRequest', pointer: '/record', imports: ['./stored-quote.js'], accepted: ['stored-quote.js'], tokens: 3072 }, reducer: { filename: 'request-reducer.js', name: 'applyRequest', checker: 'check-request-reducer.mjs', section: '## applyRequest', next: null, pointer: '/record', imports: ['./request-store.js'], accepted: ['request-store.js', 'stored-quote.js'], tokens: 4096 } }[unit];
assert.ok(config);
const out = root + '/unit-' + unit;
await mkdir(out);
const state = async (status, extra = {}) => { await writeFile(out + '/state.json', JSON.stringify({ status, pid: process.pid, at: new Date().toISOString(), ...extra }, null, 2)); console.log(status); };
try {
  const sourceHashes = await sourceBinding(target), harnessHashes = await harnessBinding(), driver = await readFile(new URL(import.meta.url)), checker = await readFile(root + '/' + config.checker), fullContract = await readFile(root + '/STORED-DATA-CONTRACT.md', 'utf8');
  assert.equal(sourceHashes[config.filename], undefined);
  const contract = fullContract.slice(0, fullContract.indexOf('## isValidStoredQuote')) + fullContract.slice(fullContract.indexOf(config.section), config.next ? fullContract.indexOf(config.next) : undefined);
  const guidePath = 'skills/qwen-samkill-ui/references/decomposed-layout.md', fullGuide = await readFile(guidePath, 'utf8'), guide = fullGuide.slice(fullGuide.indexOf('## Behavior units'), fullGuide.indexOf('## Executable layout'));
  const data = await readElementDataContext(target, { dataSources: [{ id: 'actual-' + unit, file: 'booking-fixtures.json', pointer: config.pointer }] });
  const accepted = await Promise.all(config.accepted.map(async file => file + '\n' + await readFile(target + '/' + file, 'utf8')));
  const request = { model: 'qwen3.6:35b-a3b-coding', stream: false, think: false, options: { num_ctx: 16384, num_gpu: 16, num_batch: 128, num_predict: config.tokens, temperature: .7, top_p: .8, top_k: 20, min_p: 0, presence_penalty: 1.5, repeat_penalty: 1 }, format: { type: 'object', properties: { code: { type: 'string' } }, required: ['code'], additionalProperties: false }, messages: [{ role: 'system', content: 'Implement exactly one bounded pure JavaScript ESM module. Return JSON {"code":"complete module"}. Export only ' + config.name + '. Only allowed import files: ' + config.imports.join(',') + '. No comments, Markdown, XML, DOM, storage, network, ambient clock or side effects. Never throw on malformed JSON-shaped inputs; return the contracted invalid result. Preserve all input values. Implement every exact-key/type/arithmetic/identity guard. No other module or UI.' }, { role: 'user', content: 'APPLICABLE BOUNDED BEHAVIOR GUIDE\n' + guide + '\nONLY WRITABLE CONTRACT\n' + contract + '\nREAD-ONLY ACCEPTED IMPORT IMPLEMENTATIONS\n' + accepted.join('\n\n') + '\nACTUAL SOURCE DATA\n' + data.text }] };
  await writeFile(out + '/driver.mjs.txt', driver);
  await writeFile(out + '/request.json', JSON.stringify({ request, sourceHashes, harnessHashes, contractSha256: hash(fullContract), checkerSha256: hash(checker), driverSha256: hash(driver), guideSource: { path: guidePath, sourceSha256: hash(fullGuide), deliveredSha256: hash(guide) }, dataSources: data.sources, assistance: 'Host frozen typed interface, actual fixture, accepted imports, independent acceptance tests and source assembly. Actual local Qwen supplies only this bounded pure function. Not all-Qwen product or reference parity.' }, null, 2));
  await run('pwsh.exe', ['-NoProfile', '-Command', `$c=@(Get-NetTCPConnection -State Established -ErrorAction SilentlyContinue | Where-Object {$_.RemotePort -eq 11434 -and $_.OwningProcess -ne ${process.pid}});$w=@(Get-CimInstance Win32_Process | Where-Object {$_.Name -in @('node.exe','python.exe','bun.exe') -and $_.ProcessId -ne ${process.pid} -and $_.CommandLine -match '(generate-service|generate-components|repair-component|(focus-timer|stay-booking).*(run-|repair-).*mjs)'});if($c.Count -or $w.Count){exit 9}`], { windowsHide: true, timeout: 15000 });
  const ps = await inferenceFetch('http://127.0.0.1:11434/api/ps');
  await writeFile(out + '/preflight.json', JSON.stringify({ at: new Date().toISOString(), competingConnectionAndWorkerCheck: 'CLEAR', ollama: await ps.json() }, null, 2));
  await state('LOCAL_' + unit.toUpperCase() + '_ACTIVE');
  const started = Date.now(), response = await inferenceFetch('http://127.0.0.1:11434/api/chat', { method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify(request), signal: AbortSignal.timeout(600000) });
  assert.equal(response.ok, true);
  const raw = await response.json();
  await writeFile(out + '/response.json', JSON.stringify({ elapsedMs: Date.now() - started, ...raw }, null, 2));
  assert.equal(raw.done_reason, 'stop');
  const original = JSON.parse(raw.message.content).code;
  await writeFile(out + '/raw-module.js.txt', original);
  const code = removeComments('', '', original).js, ast = parse(code, { ecmaVersion: 'latest', sourceType: 'module' });
  assert.deepEqual(parse(original, { ecmaVersion: 'latest', sourceType: 'module' }), ast);
  for (const node of ast.body) if (node.type === 'ImportDeclaration') assert.ok(config.imports.includes(node.source.value));
  assert.ok(!ast.body.some(node => ['ExportDefaultDeclaration', 'ExportAllDeclaration'].includes(node.type)));
  assert.ok(!/\b(?:document|window|localStorage|fetch|process|require|eval)\b|Date\.now/.test(code));
  assert.deepEqual(await sourceBinding(target), sourceHashes);
  assert.deepEqual(await harnessBinding(), harnessHashes);
  assert.equal(hash(await readFile(root + '/' + config.checker)), hash(checker));
  assert.equal(hash(await readFile(root + '/STORED-DATA-CONTRACT.md')), hash(fullContract));
  assert.equal(hash(await readFile(guidePath)), hash(fullGuide));
  assert.equal(hash(await readFile(new URL(import.meta.url))), hash(driver));
  assert.deepEqual((await readElementDataContext(target, { dataSources: [{ id: 'actual-' + unit, file: 'booking-fixtures.json', pointer: config.pointer }] })).sources, data.sources);
  await writeFile(target + '/' + config.filename, code, { flag: 'wx' });
  await writeFile(out + '/module.js.txt', code);
  let exitCode = 0, log;
  try { const checked = await run(process.execPath, [root + '/' + config.checker, target, out + '/checks'], { windowsHide: true, timeout: 30000 }); log = checked.stdout + checked.stderr; } catch (error) { exitCode = error.code || 1; log = error.stdout + error.stderr; }
  await writeFile(out + '/check.log', log);

  for (const [file, value] of Object.entries(sourceHashes)) assert.equal(after[file], value);
  assert.deepEqual(await harnessBinding(), harnessHashes);
  const result = { status: exitCode === 0 ? 'SCOPED_DEPENDENCY_ACCEPTED_NOT_PRODUCT_APPROVAL' : 'FAILED_DEPENDENCY_PRESERVED', exitCode, moduleSha256: hash(code), sourceHashes: after, harnessHashes, checkerSha256: hash(checker), contractSha256: hash(fullContract), driverSha256: hash(driver) };
  await writeFile(out + '/result.json', JSON.stringify(result, null, 2));
  if (exitCode === 0) await writeFile(out + '/accepted.json', JSON.stringify(result, null, 2));
  await state(result.status, { exitCode });
  process.exitCode = exitCode;
} catch (error) { await state('FAILED_REQUEST_OR_VALIDATION', { error: String(error.message).replaceAll(process.cwd(), 'REPOSITORY').slice(0, 1500) }); process.exitCode = 1; }
