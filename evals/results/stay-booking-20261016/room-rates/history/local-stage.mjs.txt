const run=promisify(execFile),hash=value=>createHash('sha256').update(value).digest('hex');
export async function localStage({target,out,request,inputs,assistance,accept}){
  await mkdir(out);const state=async(status,extra={})=>{console.log(status);await writeFile(out+'/state.json',JSON.stringify({status,pid:process.pid,at:new Date().toISOString(),...extra},null,2))};
  try{const sourceHashes=await sourceBinding(target),harnessHashes=await harnessBinding(),inputHashes={};for(const p of [...inputs,'runs/stay-booking-transfer-20261016/rooms/local-stage.mjs'])inputHashes[p]=hash(await readFile(p));await writeFile(out+'/request.json',JSON.stringify({request,sourceHashes,harnessHashes,inputHashes,assistance},null,2));await run('pwsh.exe',['-NoProfile','-Command',`$c=@(Get-NetTCPConnection -State Established -ErrorAction SilentlyContinue | Where-Object {$_.RemotePort -eq 11434 -and $_.OwningProcess -ne ${process.pid}});$w=@(Get-CimInstance Win32_Process | Where-Object {$_.Name -in @('node.exe','python.exe','bun.exe') -and $_.ProcessId -ne ${process.pid} -and $_.CommandLine -match '(generate-service|generate-components|repair-component|(focus-timer|stay-booking).*(run-|repair-).*mjs)'});if($c.Count -or $w.Count){exit 9}`],{windowsHide:true,timeout:15000});await state('LOCAL_QWEN_ACTIVE');const start=Date.now(),r=await inferenceFetch('http://127.0.0.1:11434/api/chat',{method:'POST',headers:{'content-type':'application/json'},body:JSON.stringify(request),signal:AbortSignal.timeout(600000)});assert.ok(r.ok);const raw=await r.json();await writeFile(out+'/response.json',JSON.stringify({elapsedMs:Date.now()-start,...raw},null,2));assert.equal(raw.done_reason,'stop');assert.deepEqual(await sourceBinding(target),sourceHashes);assert.deepEqual(await harnessBinding(),harnessHashes);for(const[p,v]of Object.entries(inputHashes))assert.equal(hash(await readFile(p)),v);const result=await accept(JSON.parse(raw.message.content),raw);await writeFile(out+'/result.json',JSON.stringify({...result,sourceHashes:await sourceBinding(target),harnessHashes,inputHashes},null,2));await state(result.status,{exitCode:result.exitCode||0});if(result.exitCode)process.exitCode=result.exitCode;return result}catch(e){await state('FAILED_STAGE',{error:String(e.message).replaceAll(process.cwd(),'REPOSITORY').slice(0,1500)});process.exitCode=1;throw e}
}
